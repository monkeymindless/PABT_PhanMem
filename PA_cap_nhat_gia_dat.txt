def chuan_hoa_chu(s: str) -> str:
    """Xóa dấu tiếng Việt để so khớp không phân biệt hoa thường và dấu."""
    s = s.lower().strip()
    bang = {
        'à':'a', 'á':'a', 'ả':'a', 'ã':'a', 'ạ':'a', 'ă':'a', 'ằ':'a', 'ắ':'a', 'ẳ':'a', 'ẵ':'a', 'ặ':'a',
        'â':'a', 'ầ':'a', 'ấ':'a', 'ẩ':'a', 'ẫ':'a', 'ậ':'a', 'đ':'d', 'è':'e', 'é':'e', 'ẻ':'e', 'ẽ':'e',
        'ẹ':'e', 'ê':'e', 'ề':'e', 'ế':'e', 'ể':'e', 'ễ':'e', 'ệ':'e', 'ì':'i', 'í':'i', 'ỉ':'i', 'ĩ':'i',
        'ị':'i', 'ò':'o', 'ó':'o', 'ỏ':'o', 'õ':'o', 'ọ':'o', 'ô':'o', 'ồ':'o', 'ố':'o', 'ổ':'o', 'ỗ':'o',
        'ộ':'o', 'ơ':'o', 'ờ':'o', 'ớ':'o', 'ở':'o', 'ỡ':'o', 'ợ':'o', 'ù':'u', 'ú':'u', 'ủ':'u', 'ũ':'u',
        'ụ':'u', 'ư':'u', 'ừ':'u', 'ứ':'u', 'ử':'u', 'ữ':'u', 'ự':'u', 'ỳ':'y', 'ý':'y', 'ỷ':'y', 'ỹ':'y', 'ỵ':'y'
    }
    return "".join(bang.get(c, c) for c in s)

# GIẢI PHÁP 1: TẢI & LỌC TỰ ĐỘNG TỪ WEB / INTERNET
# ------------------------------------------------------------------------------
def tai_va_loc_tu_web(tinh_nhap: str, xa_nhap: str) -> Tuple[str, str, List[Dict[str, Any]]]:
    """Tải CSDL Bảng giá đất toàn tỉnh từ Web và tự động lọc đúng xã/phường."""
    tinh_kd = chuan_hoa_chu(tinh_nhap)
    xa_kd = chuan_hoa_chu(xa_nhap)

    # Nhận diện tỉnh
    ten_tinh_chuan = "Đồng Nai"
    if "binh duong" in tinh_kd: ten_tinh_chuan = "Bình Dương"
    elif any(k in tinh_kd for k in ["hcm", "ho chi minh", "sai gon"]): ten_tinh_chuan = "TP. Hồ Chí Minh"
    elif "long an" in tinh_kd: ten_tinh_chuan = "Long An"
    elif "ha noi" in tinh_kd: ten_tinh_chuan = "Hà Nội"
    else: ten_tinh_chuan = tinh_nhap.strip().title()

    url_tinh = f"{URL_KHO_DU_LIEU_TINH}/{urllib.parse.quote(ten_tinh_chuan)}/toan_tinh.json"
    req = urllib.request.Request(url_tinh, headers={"User-Agent": "PABT-Enterprise/6.0"})
    
    # Kết nối web với timeout 5 giây
    with urllib.request.urlopen(req, timeout=5) as resp:
        ds_toan_tinh = json.loads(resp.read().decode("utf-8"))

    ds_loc = []
    ten_xa_chuan = xa_nhap.strip()
    for row in ds_toan_tinh:
        xa_goc = row.get("xa", "")
        if xa_kd in chuan_hoa_chu(xa_goc) or chuan_hoa_chu(xa_goc) in xa_kd:
            ten_xa_chuan = xa_goc
            item = dict(row)
            item["tinh"] = ten_tinh_chuan
            item["can_cu"] = f"Bảng giá đất tỉnh {ten_tinh_chuan} năm 2026"
            ds_loc.append(item)

    if not ds_loc:
        raise ValueError(f"Không tìm thấy dữ liệu giá đất của '{xa_nhap}' trên kho trực tuyến của tỉnh {ten_tinh_chuan}.")

    return (ten_tinh_chuan, ten_xa_chuan, ds_loc)

# GIẢI PHÁP 2: BÓC TÁCH TỰ ĐỘNG TỪ FILE EXCEL, WORD, PDF CỤC BỘ
# ------------------------------------------------------------------------------
def doc_va_loc_tu_file(duong_dan_file: str, tinh: str, xa_nhap: str) -> Tuple[str, str, List[Dict[str, Any]]]:
    """Tự động đọc tệp Excel/Word/PDF và lọc ra các tuyến đường thuộc xã/phường."""
    p = Path(duong_dan_file)
    duoi = p.suffix.lower()
    cac_dong: List[List[str]] = []

    # 1. Đọc Excel
    if duoi in (".xlsx", ".xlsm"):
        import openpyxl
        with open(p, "rb") as f:
            wb = openpyxl.load_workbook(f, read_only=True, data_only=True)
            try:
                for ws in wb.worksheets:
                    for r in ws.iter_rows(values_only=True):
                        dong_chu = [("" if v is None else str(v)).strip() for v in r]
                        if any(dong_chu): cac_dong.append(dong_chu)
            finally:
                wb.close()

    # 2. Đọc Word
    elif duoi == ".docx":
        import docx
        doc = docx.Document(p)
        for table in doc.tables:
            for row in table.rows:
                dong_chu = [cell.text.strip() for cell in row.cells]
                if any(dong_chu): cac_dong.append(dong_chu)

    # 3. Đọc PDF
    elif duoi == ".pdf":
        import pypdf
        with open(p, "rb") as f:
            reader = pypdf.PdfReader(f)
            for page in reader.pages:
                text = page.extract_text() or ""
                for line in text.splitlines():
                    parts = re.split(r"\s{2,}|\t", line.strip())
                    if any(parts): cac_dong.append([x.strip() for x in parts])

    elif duoi in (".csv", ".txt"):
        txt = p.read_text(encoding="utf-8-sig", errors="replace")
        cac_dong = [[c.strip() for c in r] for r in csv.reader(txt.splitlines()) if any(r)]
    else:
        raise ValueError(f"Định dạng tệp {duoi} không được hỗ trợ!")

    if not cac_dong:
        raise ValueError(f"Không thể đọc được bảng dữ liệu từ tệp {p.name}!")

    # Bóc tách cấu trúc bảng tự động
    idx_tuyen = 1 if len(cac_dong[0]) > 1 else 0
    idx_vt1 = 3 if len(cac_dong[0]) > 3 else 2
    xa_kd = chuan_hoa_chu(xa_nhap)

    ds_loc = []
    tuyen_hien_tai = "Tuyến đường chính"

    for r in cac_dong[1:]:
        if len(r) <= idx_tuyen: continue
        o_tuyen = r[idx_tuyen].strip()
        if o_tuyen and not o_tuyen.isdigit() and len(o_tuyen) > 3:
            tuyen_hien_tai = o_tuyen

        cot_gia = []
        for c in range(idx_vt1, min(idx_vt1 + 4, len(r))):
            clean = re.sub(r"[^\d]", "", r[c])
            if clean: cot_gia.append(clean)

        if cot_gia:
            for vt_num, g in enumerate(cot_gia, start=1):
                ds_loc.append({
                    "tinh": tinh or "Đồng Nai",
                    "xa": xa_nhap.strip().title(),
                    "tuyen_duong": tuyen_hien_tai,
                    "loai_dat": "ODT",
                    "vi_tri": f"Vị trí {vt_num}",
                    "gia": g,
                    "he_so_k": "1.0",
                    "can_cu": f"Trích xuất tự động từ {p.name}"
                })

    if not ds_loc:
        raise ValueError(f"Không nhận diện được mức đơn giá nào cho '{xa_nhap}' trong tệp {p.name}!")

    return (tinh or "Đồng Nai", xa_nhap.strip().title(), ds_loc)


# ------------------------------------------------------------------------------
# HÀM NẠP CSDL SQLITE (GHI ĐÈ & XÓA SẠCH 4 XÃ CŨ)
# ------------------------------------------------------------------------------
def ghi_gia_dat_vao_csdl(con: sqlite3.Connection, ds_gia_dat: List[Dict[str, Any]], xoa_cu: bool = True) -> int:
    cur = con.cursor()
    cur.execute("PRAGMA table_info(gia_dat)")
    cot_hien_tai = {r[1] for r in cur.fetchall()}
    for c in ("tinh_thanh", "quan_huyen"):
        if c not in cot_hien_tai:
            try: cur.execute(f"ALTER TABLE gia_dat ADD COLUMN {c} TEXT DEFAULT ''")
            except Exception: pass

    # Xóa sạch 4 xã cũ (Trị An, Tân An, Phú Lý, Tân Triều)
    if xoa_cu:
        cur.execute("DELETE FROM gia_dat")

    so_dong = 0
    for d in ds_gia_dat:
        cur.execute("""
            INSERT INTO gia_dat (xa_phuong, loai_dat, vi_tri, tuyen_duong, gia, he_so_k, can_cu, tinh_thanh, quan_huyen)
            VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)
        """, (
            d.get("xa", ""), d.get("loai_dat", "ODT"), d.get("vi_tri", "Vị trí 1"),
            d.get("tuyen_duong", ""), str(d.get("gia", "0")), str(d.get("he_so_k", "1.0")),
            d.get("can_cu", "Bảng giá đất 2026"), d.get("tinh", ""), d.get("huyen", "")
        ))
        so_dong += 1

    con.commit()
    return so_dong

        tinh = self.ed_tinh.text().strip()

            self._hien_thi_va_ghi_csdl(ds, tinh_c, xa_c, nguon="Web / Internet")

                f"BẠN CÓ MUỐN CHUYỂN SANG GIẢI PHÁP 2:\n"

                QMessageBox.Yes | QMessageBox.No
            )

                self._xu_ly_giai_phap_2_file()

    def _xu_ly_giai_phap_2_file(self):
        xa = self.ed_xa.text().strip()
        tinh = self.ed_tinh.text().strip()

        file, _ = QFileDialog.getOpenFileName(
            self, f"Chọn file Bảng giá đất để tự động phân tích cho {xa}", "",
            "Tệp văn bản (*.xlsx *.docx *.pdf *.csv);;Excel (*.xlsx);;Word (*.docx);;PDF (*.pdf)"
        )
        if not file:
            return

        self.lb_status.setText(f"[GP 2] Đang tự động phân tích tệp {Path(file).name}...")
        try:
            tinh_c, xa_c, ds = doc_va_loc_tu_file(file, tinh, xa)
            self._hien_thi_va_ghi_csdl(ds, tinh_c, xa_c, nguon=f"Tệp {Path(file).name}")
        except Exception as e:
            QMessageBox.critical(self, "Lỗi phân tích file", f"Không thể bóc tách dữ liệu từ file: {e}")
            self.lb_status.setText("Phân tích file thất bại.")

            g_fmt = f"{int(g_str):,}" if str(g_str).isdigit() else str(g_str)
            self.bang_kq.setItem(i, 4, QTableWidgetItem(g_fmt))

        so_dong = ghi_gia_dat_vao_csdl(self.con, ds, xoa_cu=True)

            f"ĐÃ HOÀN TẤT CẬP NHẬT BẢNG GIÁ ĐẤT!\n\n"
            f"• Nguồn dữ liệu: {nguon}\n"

            f"• Đã xóa sạch toàn bộ 4 xã cũ (Trị An, Tân An, Phú Lý, Tân Triều).\n\n"
